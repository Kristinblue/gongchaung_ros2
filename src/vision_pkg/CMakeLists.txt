cmake_minimum_required(VERSION 3.8)
project(vision_pkg)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(robot_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(OpenCV REQUIRED)

add_executable(camera_node src/camera_node.cpp)
target_include_directories(camera_node PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(camera_node ${OpenCV_LIBS})
ament_target_dependencies(camera_node rclcpp sensor_msgs)
add_executable(safety_detector_node src/safety_detector_node.cpp src/safety_detector.cpp)
target_include_directories(safety_detector_node PRIVATE include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(safety_detector_node ${OpenCV_LIBS})
ament_target_dependencies(safety_detector_node rclcpp robot_interfaces sensor_msgs std_srvs)
install(TARGETS camera_node safety_detector_node DESTINATION lib/${PROJECT_NAME})

ament_python_install_package(vision_pkg PACKAGE_DIR modules)
foreach(node inference_node perception_node)
  install(PROGRAMS scripts/${node}.py DESTINATION lib/${PROJECT_NAME} RENAME ${node})
endforeach()
ament_package()
