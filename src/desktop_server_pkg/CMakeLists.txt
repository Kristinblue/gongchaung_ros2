cmake_minimum_required(VERSION 3.8)
project(desktop_server_pkg)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(robot_interfaces REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Threads REQUIRED)
add_executable(desktop_server_node src/desktop_server_node.cpp)
target_include_directories(desktop_server_node PRIVATE third_party ${OpenCV_INCLUDE_DIRS})
target_link_libraries(desktop_server_node ${OpenCV_LIBS} Threads::Threads)
ament_target_dependencies(desktop_server_node ament_index_cpp rclcpp sensor_msgs rcl_interfaces robot_interfaces)
install(TARGETS desktop_server_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY web DESTINATION share/${PROJECT_NAME})
install(FILES third_party/LICENSE.cpp-httplib DESTINATION share/${PROJECT_NAME})
ament_package()
