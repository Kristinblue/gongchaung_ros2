cmake_minimum_required(VERSION 3.16)
project(GongchuangClient LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
set(OpenCV_STATIC ON)
find_package(OpenCV 4.5.4 REQUIRED COMPONENTS core imgproc)
set(vision_dir ${CMAKE_CURRENT_SOURCE_DIR}/../../src/vision_pkg)
add_executable(GongchuangClient WIN32 main.cpp safety_tuning.cpp ${vision_dir}/src/safety_detector.cpp)
target_include_directories(GongchuangClient PRIVATE ${vision_dir}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(GongchuangClient PRIVATE Qt6::Widgets Qt6::Network ${OpenCV_LIBS})
