cmake_minimum_required(VERSION 3.8)
project(action_pkg)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(communication_pkg REQUIRED)
find_package(vision_pkg REQUIRED)
add_executable(task_node src/task_node.cpp)
ament_target_dependencies(task_node rclcpp std_msgs std_srvs communication_pkg vision_pkg)
install(TARGETS task_node DESTINATION lib/${PROJECT_NAME})
ament_package()
