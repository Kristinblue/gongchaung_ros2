cmake_minimum_required(VERSION 3.8)
project(vision_pkg)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(communication_pkg REQUIRED)
find_package(OpenCV REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/Object.msg" "msg/Objects.msg" "msg/Region.msg" "msg/SafetyRegions.msg" DEPENDENCIES std_msgs geometry_msgs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)

foreach(node camera_node perception_node)
  add_executable(${node} src/${node}.cpp)
  target_include_directories(${node} PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(${node} ${OpenCV_LIBS} "${typesupport}")
  ament_target_dependencies(${node} rclcpp std_msgs sensor_msgs communication_pkg)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(PROGRAMS scripts/yolo_node DESTINATION lib/${PROJECT_NAME})
# 生成的 vision_pkg 消息已占用同名 Python 包。
ament_python_install_package(gongchuang_yolo)
ament_export_dependencies(rosidl_default_runtime)
ament_package()
