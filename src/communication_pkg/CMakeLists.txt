cmake_minimum_required(VERSION 3.8)
project(communication_pkg)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/Command.msg" "msg/Feedback.msg" "msg/State.msg" DEPENDENCIES builtin_interfaces)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
add_executable(serial_node src/serial_node.cpp)
ament_target_dependencies(serial_node rclcpp)
target_link_libraries(serial_node "${typesupport}")
install(TARGETS serial_node DESTINATION lib/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
